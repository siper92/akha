
Grammar of akha
===============

Script = {Line} eof .

Line = [Statement] newline .
# NOTE: newline is optional before eof

Block = '{' newline {Line} '}' .
# NOTE: '{' ends the header line, '}' starts its own line

Statement = LetStmt
          | VarStmt
          | IfStmt
          | ForStmt
          | BreakStmt
          | ContinueStmt
          | ReturnStmt
          | AssignStmt
          | CallStmt
          .

LetStmt = 'let' identifier '=' Expr .

VarStmt = 'var' identifier ['=' Expr] .
# NOTE: '_' is only a LetStmt name

AssignStmt = Target '=' Expr .

CallStmt = PostfixExpr .
# NOTE: the last suffix of a CallStmt is a CallSuffix

Target = identifier {DotSuffix | IndexSuffix} .
# NOTE: '_' is not a Target

IfStmt = 'if' Header Block ['else' (IfStmt | Block)] .
# NOTE: 'else' sits on the '}' line

ForStmt = 'for' ['var'] LoopVariables ('in' Header | 'range' Range) Block .

LoopVariables = identifier [',' identifier] .
# NOTE: 'range' takes one loop variable

Range = '[' Expr '..' Expr ']' .

BreakStmt = 'break' .

ContinueStmt = 'continue' .
# NOTE: BreakStmt and ContinueStmt only appear inside a ForStmt Block

ReturnStmt = ('return' | 'exit') [Expr] .

Header = Expr .
# NOTE: a Header cannot start with '{', wrap an object in '(' ')'

Expr = OrExpr .

OrExpr = AndExpr {('or' | '||') AndExpr} .

AndExpr = NotExpr {('and' | '&&') NotExpr} .

NotExpr = ('not' | '!') NotExpr | InExpr .

InExpr = CmpExpr [('in' | 'not' 'in') CmpExpr] .
# NOTE: InExpr does not chain

CmpExpr = AddExpr [('==' | '!=' | '<' | '<=' | '>' | '>=') AddExpr] .
# NOTE: CmpExpr does not chain

AddExpr = MulExpr {('+' | '-') MulExpr} .

MulExpr = UnaryExpr {('*' | '/' | '%') UnaryExpr} .

UnaryExpr = '-' UnaryExpr | PostfixExpr .

PostfixExpr = Operand {DotSuffix | IndexSuffix | CallSuffix} .

DotSuffix = '.' identifier .

IndexSuffix = '[' Expr ']' .
# NOTE: a slice 'a[1:2]' is an error

CallSuffix = '(' [List] ')' .

List = Expr {',' Expr} [','] .

Operand = identifier
        | number
        | String
        | 'true' | 'false' | 'null'
        | ArrayExpr
        | ObjectExpr
        | '(' Expr ')'
        .

ArrayExpr = '[' [List] ']' .

ObjectExpr = '{' [Entry {',' Entry} [',']] '}' .

Entry = (identifier | string) ':' Expr .
# NOTE: keys are unique inside one ObjectExpr

String = string | Template .

Template = '"' {char | escape | '${' ValueRef '}'} '"' .

ValueRef = identifier {DotSuffix | '[' (number | string | ValueRef) ']'} .


# Lexical grammar
identifier = letter {letter | digit} .
letter     = 'a' ... 'z' | 'A' ... 'Z' | '_' .
digit      = '0' ... '9' .
nonzero    = '1' ... '9' .
number     = ('0' | nonzero {digit}) ['.' digit {digit}] .
string     = '"' {char | escape} '"' .
escape     = '\' ('n' | 't' | 'r' | '"' | '\' | '$') .
comment    = '//' {char} .


# Notation (similar to Go spec and starlark grammar.txt)
- lowercase and 'quoted' items are lexical tokens
- Capitalized names denote grammar productions
- (...) implies grouping
- x | y means either x or y
- [x] means x is optional
- {x} means x is repeated zero or more times
- the end of each declaration is marked with a period

# Tokens
- spaces: newline, eof
- identifier, number, string
- plus all quoted tokens such as '..', 'return'
- keywords are case insensitive: let var if else for in range break continue return exit and or not true false null
- reserved words are case insensitive: fn try catch
- identifiers are case sensitive and are not keywords or reserved words
- aliases: '&&' is 'and', '||' is 'or', '!' is 'not'

# Notes
- spaces, tabs and comments are skipped, newline ends a statement
- newlines are ignored inside '(' ')', '[' ']' and an ObjectExpr
- ambiguity is resolved with the precedence of the Expr productions, lowest first
- the parser rejects nesting deeper than 1000 levels
- every Capitalized production has a matching parse function in parser/parser.go
